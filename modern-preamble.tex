% Use installed Windows fonts by file path so XeLaTeX does not depend on the
% host's fontconfig cache. On other systems, Pandoc's TeX defaults are used.
\usepackage{fontspec}

% Colors
\usepackage[dvipsnames]{xcolor}
\definecolor{DeepBlue}{HTML}{2e4a62}
\definecolor{LinkBlue}{HTML}{0078d4}

\IfFileExists{C:/Windows/Fonts/georgia.ttf}{
  \setmainfont{Georgia}[
    Path=C:/Windows/Fonts/,
    Extension=.ttf,
    UprightFont=georgia,
    BoldFont=georgiab,
    ItalicFont=georgiai,
    BoldItalicFont=georgiaz
  ]
}{}
\IfFileExists{C:/Windows/Fonts/arial.ttf}{
  \setsansfont{Arial}[
    Path=C:/Windows/Fonts/,
    Extension=.ttf,
    UprightFont=arial,
    BoldFont=arialbd,
    ItalicFont=ariali,
    BoldItalicFont=arialbi
  ]
}{}
\IfFileExists{C:/Windows/Fonts/consola.ttf}{
  \setmonofont{Consolas}[
    Path=C:/Windows/Fonts/,
    Extension=.ttf,
    UprightFont=consola,
    BoldFont=consolab,
    ItalicFont=consolai,
    BoldItalicFont=consolaz
  ]
}{}

% Section Header Styling
\usepackage{titlesec}
\newcommand{\sectionbreak}{\clearpage}
\titleformat{\section}{\color{DeepBlue}\Large\bfseries\sffamily}{\thesection}{1em}{}
\titleformat{\subsection}{\color{DeepBlue}\large\bfseries\sffamily}{\thesubsection}{1em}{}
\titleformat{\subsubsection}{\color{DeepBlue}\normalsize\bfseries\sffamily}{\thesubsubsection}{1em}{}

% Captions
\usepackage{caption}
\DeclareCaptionFont{deepblue}{\color{DeepBlue}}
\captionsetup{font=small,labelfont={bf,deepblue},textfont=it}

% Page Layout
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\renewcommand{\headrulewidth}{0.4pt}
\renewcommand{\footrulewidth}{0pt}
\fancyhead[L]{\small\color{gray}\sffamily\leftmark}
\fancyhead[R]{\small\color{gray}\thepage}

% Links
\usepackage{hyperref}
\hypersetup{
    colorlinks=true,
    linkcolor=LinkBlue,
    filecolor=LinkBlue,      
    urlcolor=LinkBlue,
    citecolor=LinkBlue,
}

% Tables
\usepackage{booktabs}
